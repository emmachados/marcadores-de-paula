cmake_minimum_required(VERSION 3.22)
set(QT_PARENT_PROJECT_NAME "QtBase")
set(QT_STANDALONE_PROJECT_NAME "PatchedCocoaPlugin")
project(PatchedCocoaPlugin VERSION 6.12.0 LANGUAGES C CXX OBJC OBJCXX)
find_package(Qt6 6.12.0 EXACT CONFIG REQUIRED COMPONENTS BuildInternals Core CorePrivate Gui GuiPrivate)
qt_build_repo_begin()
foreach(name Foundation AppKit Carbon CoreServices CoreVideo IOKit IOSurface Metal QuartzCore UniformTypeIdentifiers)
  find_library(FW${name} ${name} REQUIRED)
endforeach()
if(NOT DEFINED QTBASE_SOURCE_DIR)
  message(FATAL_ERROR "QTBASE_SOURCE_DIR must name the verified, patched Qt source tree")
endif()
add_subdirectory(${QTBASE_SOURCE_DIR}/src/plugins/platforms/cocoa cocoa)
qt_build_repo_end()
